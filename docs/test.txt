1. Installation des dépendances

2. Script complet index_chroma_hf.py
python
import json
import os
from typing import List, Dict

from langchain_community.embeddings import HuggingFaceEmbeddings
from langchain_chroma import Chroma

from metadata_utils import normalize_metadata_for_chroma

# --------------------
# CONFIG
# --------------------
ISSUES_PATH = "/home/test-pc/Bureau/PFE_Chatbot/PFE_Chatbot_STM32Cube/data/chunks_issues_stm32cubeh7_v2.json"
FILES_PATH = "/home/test-pc/Bureau/PFE_Chatbot/PFE_Chatbot_STM32Cube/data/chunks_files_stm32cubeh7_v2.json"

PERSIST_DIR = "./chroma_db_hf"  # répertoire pour cette version
MAX_CHUNK_SIZE = 1000  # limite de texte pour les embeddings


def load_json(path: str) -> List[Dict]:
    if not os.path.exists(path):
        raise FileNotFoundError(f"File not found: {path}")
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)


def main():
    print("=== START INDEXING (HuggingFaceEmbeddings) ===")

    print("Checking paths...")
    print("Issues exists:", os.path.exists(ISSUES_PATH))
    print("Files exists:", os.path.exists(FILES_PATH))

    try:
        issues = load_json(ISSUES_PATH)
        print("Issues loaded:", len(issues))
    except Exception as e:
        print("Issues error:", e)
        return

    try:
        files = load_json(FILES_PATH)
        print("Files loaded:", len(files))
    except Exception as e:
        print("Files error:", e)
        return

    all_chunks = issues + files
    print("Total chunks:", len(all_chunks))

    if not all_chunks:
        print("No data found → STOP")
        return

    # --------------------
    # Préparation textes + metadata
    # --------------------
    print("Preparing texts and metadata...")

    texts = [
        (c.get("text") or "")[:MAX_CHUNK_SIZE]
        for c in all_chunks
    ]

    metadatas = [
        normalize_metadata_for_chroma(c.get("metadata", {}))
        for c in all_chunks
    ]

    # Sanity check
    max_len = max(len(t) for t in texts)
    avg_len = sum(len(t) for t in texts) / len(texts)
    print(f"Max chunk length: {max_len}")
    print(f"Avg chunk length: {avg_len:.1f}")

    # --------------------
    # Embeddings HF
    # --------------------
    print("Creating HuggingFace embeddings model...")
    embeddings = HuggingFaceEmbeddings(
        model_name="sentence-transformers/all-MiniLM-L6-v2"
    )

    # --------------------
    # Chroma DB
    # --------------------
    print("Creating Chroma DB and adding texts...")
    db = Chroma(
        persist_directory=PERSIST_DIR,
        embedding_function=embeddings,
        collection_metadata={"hnsw:space": "cosine"}
    )

    db.add_texts(texts=texts, metadatas=metadatas)

    db.persist()
    print("DONE! Chroma DB (HF) created successfully at", PERSIST_DIR)


if __name__ == "__main__":
    main()

✅ Solution B – Ollama + nomic-embed-text avec pré calcul des embeddings
Ici, on évite les add_texts batch par batch, on fait :
1.	Calcul de tous les embeddings via embed_documents.
2.	Création de Chroma avec from_embeddings.
1. Prérequis
•	Ollama installé + modèle nomic-embed-text téléchargé :
bash
ollama pull nomic-embed-text
•	Paquets Python :
bash
pip install "langchain-ollama>=0.1.0" "langchain-chroma>=0.1.0"
2. Script complet index_chroma_ollama_nomic.py
python
import json
import os
from typing import List, Dict

from langchain_ollama import OllamaEmbeddings
from langchain_chroma import Chroma

from metadata_utils import normalize_metadata_for_chroma

# --------------------
# CONFIG
# --------------------
ISSUES_PATH = "/home/test-pc/Bureau/PFE_Chatbot/PFE_Chatbot_STM32Cube/data/chunks_issues_stm32cubeh7_v2.json"
FILES_PATH = "/home/test-pc/Bureau/PFE_Chatbot/PFE_Chatbot_STM32Cube/data/chunks_files_stm32cubeh7_v2.json"

PERSIST_DIR = "./chroma_db_nomic"  # autre répertoire pour cette version
MAX_CHUNK_SIZE = 1000


def load_json(path: str) -> List[Dict]:
    if not os.path.exists(path):
        raise FileNotFoundError(f"File not found: {path}")
    with open(path, "r", encoding="utf-8") as f:
        return json.load(f)


def main():
    print("=== START INDEXING (Ollama + nomic-embed-text) ===")

    print("Checking paths...")
    print("Issues exists:", os.path.exists(ISSUES_PATH))
    print("Files exists:", os.path.exists(FILES_PATH))

    try:
        issues = load_json(ISSUES_PATH)
        print("Issues loaded:", len(issues))
    except Exception as e:
        print("Issues error:", e)
        return

    try:
        files = load_json(FILES_PATH)
        print("Files loaded:", len(files))
    except Exception as e:
        print("Files error:", e)
        return

    all_chunks = issues + files
    print("Total chunks:", len(all_chunks))

    if not all_chunks:
        print("No data found → STOP")
        return

    # --------------------
    # Préparation textes + metadata
    # --------------------
    print("Preparing texts and metadata...")

    texts = [
        (c.get("text") or "")[:MAX_CHUNK_SIZE]
        for c in all_chunks
    ]

    metadatas = [
        normalize_metadata_for_chroma(c.get("metadata", {}))
        for c in all_chunks
    ]

    max_len = max(len(t) for t in texts)
    avg_len = sum(len(t) for t in texts) / len(texts)
    print(f"Max chunk length: {max_len}")
    print(f"Avg chunk length: {avg_len:.1f}")

    # --------------------
    # Embeddings avec Ollama (nomic-embed-text)
    # --------------------
    print("Creating OllamaEmbeddings model (nomic-embed-text)...")
    emb_model = OllamaEmbeddings(model="nomic-embed-text")

    print("Computing embeddings for all documents (this may take some time)...")
    # embed_documents prend une liste de textes
    vectors = emb_model.embed_documents(texts)

    print(f"Embeddings computed: {len(vectors)} vectors")

    # --------------------
    # Création Chroma à partir des embeddings
    # --------------------
    print("Creating Chroma DB from precomputed embeddings...")
    db = Chroma.from_embeddings(
        texts=texts,
        embeddings=vectors,
        metadatas=metadatas,
        persist_directory=PERSIST_DIR,
        collection_metadata={"hnsw:space": "cosine"}
    )

    db.persist()
    print("DONE! Chroma DB (nomic) created successfully at", PERSIST_DIR)


if __name__ == "__main__":
    main()

1. C’est quoi la différence entre ton code et un pipeline RAG “pro” (comme l’outil ST) ?
Tu es dans cette situation :
•	PC perso, probablement CPU only, pas forcément optimisé pour l’IA.
•	Tu utilises Ollama + nomic-embed-text, un modèle d’embeddings plutôt lourd.

o	plein d’appels embeddings,

L’outil ST, lui, a généralement :
•	Une infra dimensionnée (serveurs, parfois GPU, stockage plus rapide).
•	Des modèles d’embeddings choisis / calibrés pour le use case et la prod.
•	Un pipeline RAG où :
o	l’indexation (création de l’index vectoriel) est un process offline qu’on accepte de faire durer (mais sur de bonnes machines),
o	la latence critique, c’est la phase requête (retrieval + LLM), pas l’indexation.
Et surtout, la lenteur que tu observes vient de la combinaison actuelle :
•	nomic-embed-text via Ollama,
•	sur ta machine,
•	
o	la façon dont tu l’appelles (initialement batch par batch avec Chroma).
Ce n’est pas un “problème de data STM32Cube”.

2. Est ce que l’outil ST peut avoir le même problème avec tes datasets ?
Pas forcément, pour plusieurs raisons :
2.1. Volume de data
Tu as donné les chiffres :
•	issues loaded: 849
•	files loaded: 4 535
•	total chunks: 5 385
Pour un moteur RAG en production, 5 000–10 000 chunks, c’est un petit volume.
Donc :
•	côté perfs, c’est tout à fait raisonnable,
•	ce n’est pas ton dataset qui est “trop gros”.
2.2. Choix du modèle d’embeddings et de l’infra
L’outil ST :
•	peut utiliser un autre modèle d’embeddings (plus rapide / adapté à l’infra),
•	tourne probablement sur une machine serveur (voire plusieurs), donc :
o	CPU plus musclé,
o	parfois GPU,
o	SSD rapide,
o	paramétrage optimisé.
Donc un pipeline interne ST qui :
•	prend tes chunks_*_v2.json,
•	génère des embeddings,
•	construit un index FAISS / Chroma / autre
sera souvent beaucoup plus rapide que ton sandbox sur un laptop.
2.3. Stratégie d’indexation
En interne, ils peuvent :
•	faire une indexation bulk (comme la “Solution B” que je t’ai donnée),
•	paralléliser,
•	batcher intelligemment,
•	et surtout : ils ne le font pas à chaque requête, seulement lors d’une ingestion (une fois par version du dataset).
Donc même si l’indexation prend, disons, 10–20 minutes pour une version de dataset, ce n’est pas un problème en prod :
c’est un job offline, pas visible pour l’utilisateur final.

Titre : Nouvelle architecture du projet
Schéma textuel à mettre (ou à transformer en diagramme) :
text
PFE_Chatbot_STM32Cube/
  pipeline/
    ingestion/      → Récupère issues + fichiers STM32Cube

  datasets/         → Jeux de données "officiels" RAG-ready (livrables)
  data/             → Sorties intermédiaires / travail
  docs/             → Doc technique, rapport, présentations

  run_full_workflow.py   → Lance tout le pipeline de bout en bout
  validate_schemas.py    → Vérifie la conformité aux schémas JSON
À dire à l’oral :
•	J’ai restructuré le projet pour qu’il soit modulaire et industrialisable.
•	pipeline/* = chaque étape logique du preprocessing.
•	shared/* = tout ce qui est commun (config, utils, schémas JSON).
•	datasets/ = les artefacts stables qui seront ingérés par le RAG ST.
•	run_full_workflow.py lance l’ensemble du pipeline, et validate_schemas.py garantit la conformité des JSON.
